# CONFIG.md — конфігураційний файл

Розташування: `%APPDATA%\AiLimits\config.toml`. Створюється автоматично
при першому запуску; кожна зміна з контекстного меню зберігається сюди.
Невідомі поля і провайдери ігноруються (старі конфіги ніколи не ламають
парсинг). При помилці парсингу віджет пише попередження в лог і бере
дефолти.

Поруч віджет може тримати `provider-cache.json` (останні відомі метрики з
майбутнім скиданням — переживають рестарт), `history.jsonl` (обмежена історія
використання для спарклайна в Expanded) і, при заданому `AILIMITS_LOG` або
`RUST_LOG`, `ailimits.log`.

# Опитування стає на паузу при бездіяльності/заблокованому екрані та

# Індикатор на панелі завдань (меню: Indicator): "tray" — трей-іконка
# 16px із двох концентричних кілець, що несуть два найнавантаженіші
# провайдери; монохромна, у темі системної панелі завдань; "panel_rows" / "panel_grid" (застарілий
# псевдонім) — прозорий overlay зліва від трею: layered-вікно з per-pixel
# alpha, малюються лише цифри й бари. Показує перші два провайдери в
# порядку віджета (відсоток + бар, розмір годинника), монохром, слідує
# СИСТЕМНІЙ світлій/темній темі; решта — у tooltip. Стежить за
# авто-приховуванням і шириною трея подієво, без полінгу. "bars" —
# застаріла трей-іконка; "off" — нічого. Замінює застарілий прапорець
# show_tray_icon (ігнорується).
indicator = "tray"
# Автооновлення (меню: Automatic updates). Коли true, AI Limits у фоні
# перевіряє GitHub на новіший реліз (~через 30с після старту, далі щодоби),
# звіряє інсталятор із опублікованим у релізі SHA-256, тихо встановлює його
# й перезапускається. false — вимкнути; відсутнє в старих конфігах → true.
auto_update = true
# Ручне зміщення позиції панелі, у пікселях, застосовується після
# обчисленого розташування. Обидва за замовчуванням 0 — без коригування.
# Обмежується діапазоном -200..200 — одруківка не може виштовхнути панель
# за межі робочого столу. Інструмент для відновлення на машинах, де
# автоматичне розташування не спрацьовує: заміна shell (StartAllBack,
# ExplorerPatcher), нетипові лейаути панелі завдань, лейаути на сенсорних
# пристроях. Навмисно лише в конфізі, без пункту меню.
panel_offset_x = 0
panel_offset_y = 0
# На яку панель завдань кріпиться міні-панель (меню: Indicator -> Display;
# це підменю з'являється лише коли існує вторинна панель завдань, і
# будується при старті, тож підключення монітора пізніше вимагає рестарту,
# щоб пункт з'явився). "primary" (дефолт), або { secondary = N }, де N
# рахує вторинні панелі завдань з 0, зліва направо за позицією монітора —
# напр. { secondary = 0 } для першої (найлівішої) вторинної панелі.
# Вторинні панелі завдань існують лише коли Windows налаштована показувати
# панель завдань на всіх дисплеях. Якщо обраний дисплей зник (монітор
# від'єднано, або цю опцію Windows вимкнено), або значення нерозпізнане,
# панель повертається на primary, а не ламає весь конфіг-файл.
panel_display = "primary"

[window]

# Прозорість фону 0.10–0.85.
opacity = 0.45
# Поверх всіх вікон.
pinned = false
# Заблокована позиція — drag вимкнено (незалежно від `pinned`).

[ui]
# Палітра: default / ocean / sunset / forest / neon / ice / rose / slate.
palette = "default"
# Насиченість палітри 0–100 (0 = грейскейл, 100 = повний колір).
saturation = 55
# Яскравість тексту і барів 20–100.
brightness = 100
# Монохромний режим (перекриває палітру).
monochrome = false
# Лейаут: "vertical" або "horizontal".
layout = "vertical"
# Рівень деталізації: "compact" / "medium" / "expanded".
detail = "compact"
# Крок ширини для розкладки з горизонтальними смужками (rows); вертикальна

width_scale = "full"
# Розташування колонок для вертикальних смужок; розкладка rows його ігнорує.
# "row" (поруч, широкий і низький) або "column" (стовпчиком, вузький і високий)
# (меню: Arrangement).
column_flow = "row"
# Прогноз burn-rate ("~Xh to limit", коли використання росте; меню: Forecast).
# Ніколи не замінює зворотний відлік до скидання — показується лише коли

show_forecast = false

# Рядки (меню: Rows and models...). Записується лише після вибору; без
# нього в кожного провайдера один рядок, як і раніше. Власний рядок
# провайдера — його id, додатковий рядок — provider.part, і є лише такі,
# кожен там, де провайдер рахує його окремо: claude.opus, claude.sonnet,

# grok.build, cursor.bot (Grok Bot). Віджет і підказка вибираються окремо;
# панель на панелі задач і іконка трею лишаються по одному на провайдера.
# Невідомі id ігноруються і зберігаються.
[ui.rows]
widget  = ["claude", "claude.opus", "codex"]
tooltip = ["claude", "claude.opus", "claude.sonnet", "codex", "grok"]

# Підписи моделі (меню: Rows and models...). Частини для кожної поверхні:
# "model", "version" (лише разом із "model"), "effort" (можна окремо).
# Порожньо — без підпису. Для них читається файл налаштувань кожного
# інструмента, тож перша галочка спершу питає ([reads] tool_settings).
# Віджет малює підпис окремим рядком в Expanded і в розкладці стовпцями, а в
# Medium — на тому місці рядка метаданих, що лишає скидання (або теж окремим
# рядком, коли це місце вужче за назву); Compact лишає
# його підказці. У вузьких місцях спершу відпадає effort, потім версія,
# потім модель; слово ніколи не ріжеться.
# source: "configured" (модель, вибрана в інструменті) або "last_used"
# (найновіша сесія Claude Code чи Codex CLI; спершу питає, [reads]

[ui.model]

# складається за одним правилом без таблиці (claude-opus-5-5 -> "Opus 5.5"),

# fire-and-forget; контекст події — в змінних оточення: AILIMITS_EVENT
# (threshold|reset|startup), AILIMITS_PROVIDER, AILIMITS_PERCENT,
# AILIMITS_RESET_AT (RFC3339, коли відомо).
# БЕЗПЕКА: ці поля навмисно редагуються ЛИШЕ у конфізі — немає шляху через

## Секрети

Конфіг ніколи не містить ключів чи токенів — лише мітки. Значення живуть
у Windows Credential Manager під service `ailimits`:

| Мітка | Призначення |
|---|---|
| `claude_api_key` | API ключ Claude (метод api_key) |
| `claude_usage_token` | ручний usage-токен підписки Claude |
| `codex_usage_token` | ручний usage-токен ChatGPT/Codex |
| `copilot_pat` | GitHub PAT (замість gh CLI) |
| `cursor_session` | власний сеанс Cursor віджета, як UTF-8 JSON: токен, його строк дії, остання спроба оновлення, чи завершив його Cursor (меню: *Sign in to Cursor...* / *Sign out of Cursor*; CLI: `ailimits-auth login cursor` / `logout cursor`) |

## Що читається і коли

Кожне читання файлу іншого інструмента необов'язкове, і програма спершу
питає:

- **Свіже встановлення нічого не читає.** Віджет і трей пишуть «Choose what
  to show», а вікно вибору відкривається само, і знайдені на ПК інструменти
  в ньому позначені (знайдені за тим, чи існує їхня папка; всередину програма
  не заглядає), а їхні власні рядки відмічені за одним запитанням, яке
  називає кожен файл, що буде прочитано, і кожен сервер. «Allow» запускає їх
  усі; «Cancel» (чи закрите вікно) нічого не читає і нічого не відмічає.
  Перша галочка будь-якого іншого інструмента один раз питає, називаючи файл
  і сервер.
- **Після оновлення з 0.7.0 чи старішої** нічого не зупиняється: увімкнені
  провайдери читають далі, як раніше (`sign_in` заповнюється ними), а одне
  вікно один раз перелічує всі файли, які читаються, і всі сервери, до яких
  програма звертається. Провайдер, якого стара версія не мала (для 0.7.0 це
  Grok і Cursor), додається вимкненим і питає, коли його позначити.
- **Конфіг, який не вдається прочитати**, відкладається як
  `config.toml.corrupt`, і програма стартує як після свіжого встановлення:
  що дозволяв утрачений файл, невідомо, тож нічого не читається, поки
  користувач не вибере знову. Порожній конфіг або конфіг без `[[providers]]`
  теж вважається свіжим встановленням.
- **Свіже встановлення ще й вимикає всіх провайдерів**, а не лише нічого не
  дозволяє: навіть старіша версія, запущена потім на тому самому конфігу,
  читає тільки дозволене.
- **Щоб припинити читання інструмента**, зніміть усі його рядки в Rows and
  models (або галочку Enabled у меню Providers). Підписи моделі перестають
  читатися, коли зняті Model, Version і Effort.
- **Де інструменти тримають свої файли**, можна змінити через `CODEX_HOME`,
  `GROK_HOME` і `COPILOT_HOME`; програма йде за ними, і запитання про це
  кажуть.
- **`ailimits-auth status`** — діагностика, яку ви запускаєте самі: вона
  перевіряє вхід кожного інструмента напряму (лише наявність і термін дії,
  токен ніколи не друкується) і не зважає на `[reads]`.

Файли й сервери кожного провайдера — у [PROVIDERS.md](PROVIDERS.md) і в
`src/consent.rs`, звідки беруть текст запитання.

Rust-структури — у `src/config/schema.rs`; цей файл є єдиним джерелом
правди для схеми.
